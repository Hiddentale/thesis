\subsection{Standard notation for paper}
\begin{definition}[Chromatic number]\label{chromatic}\cite{west}
        The smallest number of colors required to color the vertices of any graph \(G\), so that any adjacent vertices are never the same color, is called the chromatic number of  \(G\). Denoted as \(\chi(G)\).
    \end{definition}
    \begin{definition}[Circuit]\label{circuit}\cite{west}
        A \emph{circuit} is a closed walk that can't traverse the same edges twice.
    \end{definition}
    \begin{definition}[Cycle]\label{cycle}\cite{west}
        A \emph{cycle} is a circuit that can't traverse the same vertices twice.
    \end{definition}
    \begin{definition}[Generating set]
        Let \(G\) be a group and \(S\subseteq G\). If \(G\) can be constructed by group operations between elements of \(S\), then \(S\) is called a generating set of \(G\).
    \end{definition}
\subsection{Cayley graphs with small examples}
    \begin{definition}[Cayley Graphs]\label{cayley}
        Let \(G\) be a group, and let \(S\subseteq G\) be a generating set, construct the graph \(C(G,S)\) as follows:
        \begin{itemize}
            \item every \(g\in G\) is a vertex,
            \item for every ordered pair \((g, s)\), where \(s\cdot g_1 = g_2\) with \(s\in S\) and \(g_1, g_2\in G\), set an edge between \(g_1\) and \(g_2\), disallowing identical edges.
        \end{itemize}
        The resulting object, \(C(G,S)\), is then called a Cayley Graph.
    \end{definition}
    \begin{example}

\subsection{Minimal generating sets}
    \begin{definition}[Minimal generating set]
        Let \(G\) be a group with \(S\subseteq G\) a generating set. If removing any element of \(S\) results in \(S\) not being able to generate \(G\), we call \(S\) a minimal generating set.
    \end{definition}
    Note that a group can have multiple different generating sets, and they do not necessarily need to have the same amount of elements.
    \begin{example}
        Let \(G = \mathbb{Z}\) and \(S=\{2, 3\}\subseteq \mathbb{Z}\). It holds that \(\langle 3-2\rangle = \langle 1\rangle = \mathbb{Z}\), so \(S\) is a generating set. If we remove any of its elements, then it does not generate \(\mathbb{Z}\) anymore, so it is minimal. Also note that \(\{1\}\) is also a minimal generating set, with less elements than \(S\).\todo{improve}
    \end{example}

\subsection{Minimal Cayley graphs are no-lonely-color}
    \begin{definition}[Quasi-irreducible generating set]\label{quasi}
        Let \(G\) be a group, and let \(S\subseteq G\) be a generating set. We call \(S\) \emph{quasi-irreducible} if the group generated by \(S\) equals \(G\), e.g.\todo{Isn't this literally the definition of a generating set?}
        \[\langle S\rangle = G,\]
        and \(S\) can be well-ordered such that no element of \(S\) is contained in the group generated by the previous elements.
    \end{definition}
    Note that we only need one valid ordering for this to hold, as the following example shows.
    \begin{example}
        Let \(G = \mathbb{Z}\), and let \(S = \{2,3,4\}\). Since \(\langle 3-2\rangle = \langle 1\rangle = \mathbb{Z}\), also \(\langle S\rangle = \mathbb{Z}\).
        Observe that the the ordering \((2,4,3)\) fails since \(4\in \langle 2\rangle\), but \((4, 2, 3)\) works since \(2\notin\langle 4\rangle\) and \(3\notin\langle 4,2\rangle\). 
        Hence \(S\) is a quasi-irreducible generating set of \(\mathbb{Z}\).
    \end{example}
    \begin{counterexample}
        Let \(S = \{1, 2, 3\}\). Notice that \(\langle 1\rangle\) generates all the other elements, hence if \(1\) is not the last element in the ordering, the ordering is not valid. For example
        \[(1, 3, 2), \text{\ then\ } 3\in\langle 1\rangle,\]

        Furthermore, two options are left over, where \(1\) is the last element of the ordering
        \[(3, 2, 1), \text{\ then\ } 2\in\langle 1,3\rangle,\]
        \[(2, 3, 1), \text{\ then\ } 1\in\langle 2,3\rangle,\]
        which both also fail the criteria.
        Hence \(S\) is not a quasi-irreducible generating set of \(\mathbb{Z}\).
    \end{counterexample}
    \begin{definition}[No-pied-circuit graph]\label{no-pied}\todo{ask feedback}
        Let \(G\) be a graph. We call a circuit \emph{pied} if every single edge has a different color.
        If there exists an edge-coloring for \(G\) such that 
        \begin{itemize}
            \item every vertex can only have at most two edges of the same color,
            \item no circuit is pied,
        \end{itemize}
        then we call \(G\) a no-pied circuit graph.
    \end{definition}
    Observe that only \emph{one} edge-coloring has to exist for this definition to be satisfied. Furthermore, each circuit can have thousands of different colors, the only requirement is that \emph{one} of those colors appears twice.
    \begin{example}\label{no-pied-example}
        \leavevmode\par
        \begin{center}
            \begin{tikzpicture}
                \foreach \i in {1,2,3}
                    \node[vertex] (v\i) at ({90 + 120*(\i-1)} : 1.3cm) {\i};

                \foreach \i/\j/\c in {1/2/blue, 2/3/blue, 3/1/blue}
                    \draw[edge, color=\c] (v\i) -- (v\j);
            \end{tikzpicture}
        \end{center}
        Note that, every vertex has 2 edges of the color blue, and each circuit has 3 edges of the color blue.
    \end{example}
    \begin{counterexample}
        \leavevmode\par
        \begin{center}
            \begin{tikzpicture}
                \foreach \i in {1,2,3}
                    \node[vertex] (v\i) at ({90 + 120*(\i-1)} : 1.3cm) {\i};

                \foreach \i/\j/\c in {1/2/red, 2/3/blue, 3/1/blue}
                    \draw[edge, color=\c] (v\i) -- (v\j);
            \end{tikzpicture}
        \end{center}
        Observe that every circuit only contains the color red once, hence this is \emph{not} a no-pied-circuit graph.
    \end{counterexample}
    \begin{definition}[No-lonely-color graphs]\label{def:lonely}
    	A graph \(G\) is no-lonely-colored if there exists an edge-coloring such that 
        \begin{itemize}
            \item every vertex can only have at most two edges of the same color,\todo{figure out beter definition}
            \item no circuit exists with a color appearing once.
        \end{itemize}
    \end{definition}
    \begin{proposition}
        Every no-lonely-color graph is a no-pied-circuit graph.
    \end{proposition}
    \begin{proof}
        If no circuit exists with a color appearing once, then every circuit contains at least two edges of the same color.
    \end{proof}
    Hence the example given for a no-pied-circuit graph (\ref{no-pied-example}) is also a valid no-lonely-color graph. The converse is not true, as we demonstrate in the following counterexample.
