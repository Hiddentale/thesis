\begin{definition}[Chromatic number]\label{chromatic}\cite{west}
        The smallest number of colors required to color the vertices of any graph \(G\), so that any adjacent vertices are never the same color, is called the chromatic number of  \(G\). Denoted as \(\chi(G)\).
    \end{definition}
    \begin{definition}[Circuit]\label{circuit}\cite{west}
        A \emph{circuit} is a closed walk that can't traverse the same edges twice.
    \end{definition}
    \begin{definition}[Cycle]\label{cycle}\cite{west}
        A \emph{cycle} is a circuit that can't traverse the same vertices twice.
    \end{definition}

    \begin{definition}[Cayley Graphs]\label{cayley}
        Let \(G\) be a group, and let \(S\subseteq G\) be a generating set, construct the graph \(C(G,S)\) as follows:
        \begin{itemize}
            \item every \(g\in G\) is a vertex,
            \item for every ordered pair \((g, s)\), where \(s\cdot g_1 = g_2\) with \(s\in S\) and \(g_1, g_2\in G\), set an edge between \(g_1\) and \(g_2\), disallowing identical edges.
        \end{itemize}
        The resulting object, \(C(G,S)\), is then called a Cayley Graph.
    \end{definition}

\subsection{Minimal Cayley graphs are no-lonely-color}
    \begin{definition}[Quasi-irreducible generating set]\label{quasi}
        Let \(G\) be a group, and let \(S\subseteq G\) be a generating set. We call \(S\) \emph{quasi-irreducible} if the group generated by \(S\) equals \(G\), e.g.
        \[\langle S\rangle = G,\]
        and \(S\) can be well-ordered such that no element of \(S\) is contained in the group generated by the previous elements.
    \end{definition}
    \begin{example}
        Let \(S = \{2,3,4\}\), then \(3 - 2 = 1\). Since \(\langle 1\rangle = \mathbb{Z}\), also \(\langle S\rangle = \mathbb{Z}\).
        Now observe that the the ordening \(\{2,4,3\}\) fails since \(4\in \langle 2\rangle\), but \(\{4, 2, 3\}\) works since \(2\notin\langle 4\rangle\) and \(3\notin\langle 4,2\rangle\). 
        Hence \(S\) is a quasi-irreducible generating set of \(\mathbb{Z}\).
    \end{example}
    \begin{counterexample}
        Let \(S = \{1, 2, 3\}\). Notice that
        \[\{3, 2, 1\}, \text{\ then\ } 2\in\langle 1,3\rangle,\]
        \[\{3, 1, 2\}, \text{\ then\ } 1\in\langle 2,3\rangle,\]
        \[\{2, 3, 1\}, \text{\ then\ } 1\in\langle 2,3\rangle,\]
        \[\{2, 1, 3\}, \text{\ then\ } 3\in\langle 1,2\rangle,\]
        \[\{1, 3, 2\}, \text{\ then\ } 3\in\langle 1\rangle,\]
        \[\{1, 2, 3\}, \text{\ then\ } 2\in\langle 1\rangle,\]
        hence \(S\) is not a quasi-irreducible generating set of \(\mathbb{Z}\).
    \end{counterexample}

    \begin{definition}[No-lonely-color graphs]\label{def:lonely}
    	A graph \(G\) is no-lonely-colored if there exists an edge-coloring such that 
        \begin{itemize}
            \item every vertex can only have at most two edges of the same color,\todo{figure out beter definition}
            \item no circuit exists with a color appearing once.
        \end{itemize}
    \end{definition}
    \begin{proposition}
        Every no-lonely-color graph is a no-pied-circuit graph.
    \end{proposition}
    \begin{proof}
        If no circuit exists with a color appearing once, then every circuit contains at least two edges of the same color.
    \end{proof}
    Hence the example given for a no-pied-circuit graph (\ref{no-pied-example}) is also a valid no-lonely-color graph. The converse is not true.
